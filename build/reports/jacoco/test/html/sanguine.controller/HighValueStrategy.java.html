package sanguine.controller;

import java.util.List;
import sanguine.model.Card;
import sanguine.model.ReadOnlySanguineModel;

/**
 * Represents a strategy that prioritizes playing high-value cards.
 * We sort cards in descending order and plays the first valid move
 * with the highest value card available, otherwise we return null.
 */
public class HighValueStrategy implements PlayerStrategy {

  @Override
  public Move chooseMove(ReadOnlySanguineModel model) {
    List<Card> hand = model.getCurrentPlayerHand();
    int[] dims = model.getGameBoardDimensions();

    Move bestMove = null;
    int bestValue = -1;

    for (int cardIdx = 0; cardIdx < hand.size(); cardIdx++) {
      Card card = hand.get(cardIdx);

      for (int row = 0; row < dims[0]; row++) {
        for (int col = 0; col < dims[1]; col++) {
          if (model.isLegalMove(cardIdx, row, col)) {
            if (card.getValue() > bestValue) {
              bestValue = card.getValue();
              bestMove = new Move(cardIdx, row, col);
            }
          }
        }
      }
    }

    return bestMove;
  }
}
