package sanguine.controller;

import sanguine.model.ReadOnlySanguineModel;

/**
 * Represents a strategy that plays the first legal move found.
 * We iterate through all cards and cells in order until we find a valid move.
 * If we have no legal moves, then we pass the turn.
 */
public class FirstLegalMoveStrategy implements PlayerStrategy {

  @Override
  public Move chooseMove(ReadOnlySanguineModel model) {
    int[] dims = model.getGameBoardDimensions();
    int handSize = model.getCurrentPlayerHand().size();

    for (int cardIdx = 0; cardIdx < handSize; cardIdx++) {
      for (int row = 0; row < dims[0]; row++) {
        for (int col = 0; col < dims[1]; col++) {
          if (model.isLegalMove(cardIdx, row, col)) {
            return new Move(cardIdx, row, col);
          }
        }
      }
    }

    return null;
  }
}
