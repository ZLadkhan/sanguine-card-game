package sanguine.controller;

import java.util.List;
import sanguine.model.Card;
import sanguine.model.Cell;
import sanguine.model.Player;
import sanguine.model.ReadOnlySanguineModel;

/**
 * Represents a strategy that tries to maximize board control through pawn influence.
 * We prioritize moves which place pawns in more positions or convert opponent pawn influences.
 * We also consider value here, though not as a primary component.
 */
public class MaxInfluenceStrategy implements PlayerStrategy {

  @Override
  public Move chooseMove(ReadOnlySanguineModel model) {
    List<Card> hand = model.getCurrentPlayerHand();
    int[] dims = model.getGameBoardDimensions();
    Player currentPlayer = model.getCurrentPlayer();

    Move bestMove = null;
    double bestScore = Double.NEGATIVE_INFINITY;

    for (int cardIdx = 0; cardIdx < hand.size(); cardIdx++) {
      Card card = hand.get(cardIdx);

      for (int row = 0; row < dims[0]; row++) {
        for (int col = 0; col < dims[1]; col++) {
          if (model.isLegalMove(cardIdx, row, col)) {
            double score = scoreMove(model, card, row, col, currentPlayer);

            if (score > bestScore) {
              bestScore = score;
              bestMove = new Move(cardIdx, row, col);
            }
          }
        }
      }
    }

    return bestMove;
  }

  /**
   * Represents scoring a potential move based on influence spread and card value.
   */
  private double scoreMove(ReadOnlySanguineModel model, Card card,
                           int row, int col, Player player) {
    int[] dims = model.getGameBoardDimensions();
    boolean isBlue = (player == Player.BLUE);

    int influenceCount = 0;
    int conversionCount = 0;

    for (int newRow = 0; newRow < 5; newRow++) {
      for (int newCol = 0; newCol < 5; newCol++) {
        if (card.hasInfluenceAt(newRow, newCol, isBlue)) {
          int boardRow = row + newRow - 2;
          int boardCol = col + newCol - 2;

          if (boardRow >= 0 && boardRow < dims[0]
              && boardCol >= 0 && boardCol < dims[1]) {
            Cell target = model.getCellAt(boardRow, boardCol);

            if (target.getCard() != null) {
              continue;
            }

            if (!target.hasPawns()) {
              influenceCount++;
            } else if (target.getPawnOwner() != player) {
              conversionCount += 2;
            } else {
              influenceCount++;
            }
          }
        }
      }
    }

    return influenceCount + conversionCount + (card.getValue() * 0.5);
  }
}
